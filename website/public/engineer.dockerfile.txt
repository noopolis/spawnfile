RUN apt-get update && apt-get install -y --no-install-recommends bash ca-certificates curl git util-linux && rm -rf /var/lib/apt/lists/*

RUN npm install -g --prefix /usr/local --omit=dev --no-fund --no-audit @noopolis/mneme@0.1.1

RUN npm install -g --prefix /usr/local --omit=dev --no-fund --no-audit opencode-ai@1.19.6

COPY entrypoint.sh /opt/spawnfile/entrypoint.sh
RUN chmod +x /opt/spawnfile/entrypoint.sh

COPY distribution-report.json /spawnfile/spawnfile-report.json
LABEL com.spawnfile.compile_fingerprint='sf1:7f63514d34b4'
LABEL com.spawnfile.image_contract='spawnfile.image.v1'
LABEL com.spawnfile.project='engineer'

RUN mkdir -p '/var/lib/spawnfile' '/var/lib/spawnfile/instances/opencode/agent-engineer/home/.spawnfile-sessions' && printf '%s\n' 'spawnfile.volume-bootstrap.v1' > '/var/lib/spawnfile/instances/opencode/agent-engineer/home/.spawnfile-sessions/.spawnfile-volume-init' && chmod 600 '/var/lib/spawnfile/instances/opencode/agent-engineer/home/.spawnfile-sessions/.spawnfile-volume-init' && chown -R spawnfile:spawnfile '/var/lib/spawnfile'
USER spawnfile
